import json
from onecomp.kvcomp.codec.hayate.codec import (
    ADAPTIVE_HQ_DISTORTION_TABLE_VERSION,
    ADAPTIVE_HQ_OBJECTIVE,
    ADAPTIVE_HQ_POLICY_VERSION,
    ADAPTIVE_HQ_SELECTOR_CARRIER_VERSION,
    _adaptive_datafree_relative_distortion,
    _adaptive_value_distortion_for_update,
)

rows = []
for code_bits in range(3, 9):
    args = (code_bits, 2, "gaussian", 128, "auto", "ls_iter")
    base = _adaptive_datafree_relative_distortion(*args)
    all_clear = _adaptive_datafree_relative_distortion(*args, 1)
    constants = {
        "value_relative_distortion_base": base,
        "value_selector_carrier_relative_distortion_delta_per_vector": all_clear - base,
    }
    rows.append(
        {
            "code_bits": code_bits,
            "base": base,
            "all_clear": all_clear,
            "delta": all_clear - base,
            "effective": {
                str(n): _adaptive_value_distortion_for_update(constants, n)
                for n in (1, 2, 8, 128)
            },
        }
    )
print(json.dumps(rows, indent=2, sort_keys=True))
PY
